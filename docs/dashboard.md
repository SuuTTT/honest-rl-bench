<script>
(function () {
  function cell(text) {
    var td = document.createElement('td');
    td.textContent = (text === null || text === undefined) ? '—' : String(text);
    return td;
  }
  fetch('{{ "/assets/dashboard.json" | relative_url }}')
    .then(function (r) { return r.json(); })
    .then(function (d) {
      document.getElementById('generated').textContent =
        'Last updated: ' + d.generated;

      var ft = document.querySelector('#fleet-table tbody');
      d.fleet.forEach(function (b) {
        var tr = document.createElement('tr');
        if (b.reachable === false) tr.style.opacity = '0.45';
        tr.appendChild(cell(b.instance));
        tr.appendChild(cell(b.endpoint));
        tr.appendChild(cell(b.gpu));
        tr.appendChild(cell(b.util));
        tr.appendChild(cell(b.mem));
        tr.appendChild(cell(b.project));
        tr.appendChild(cell(b.note));
        ft.appendChild(tr);
      });

      var rt = document.querySelector('#runs-table tbody');
      d.runs.forEach(function (r) {
        var tr = document.createElement('tr');
        if (r.status === 'done') tr.style.background = '#eef7ee';
        if (r.status === 'running') tr.style.background = '#fff8e6';
        tr.appendChild(cell(r.project));
        tr.appendChild(cell(r.box));
        tr.appendChild(cell(r.what));
        tr.appendChild(cell(r.status));
        tr.appendChild(cell(r.progress));
        rt.appendChild(tr);
      });

      var qt = document.querySelector('#queue-table tbody');
      d.queue.forEach(function (q) {
        var tr = document.createElement('tr');
        tr.appendChild(cell(q.project));
        tr.appendChild(cell(q.what));
        tr.appendChild(cell(q.target_boxes));
        tr.appendChild(cell(q.blocked_on));
        qt.appendChild(tr);
      });
    })
    .catch(function (e) {
      document.getElementById('generated').textContent =
        'Failed to load dashboard.json: ' + e;
    });
})();
</script>
